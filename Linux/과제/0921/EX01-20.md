#include <stdio.h>

#include <stdlib.h>

#include <unistd.h>

#include <errno.h>

#include <string.h>

#include <sys/types.h>

#include <sys/stat.h>

#include <fcntl.h>

  

#define MAX_BLOCK_SIZE 4096

  

int main(int argc, char **argv)

{

    int block_size;

    off_t i, file_size, block_count;

    size_t remainder, copied = 0;

    FILE *src_fp, *dst_fp;

    char *src_name, *dst_name;

    char buf[MAX_BLOCK_SIZE];

    struct stat src_stat;

  

    if(argc != 4) {

        printf("usage: %s {src file} {dst file} {block size in bytes}\n", argv[0]);

        return EXIT_FAILURE;

    }

    printf("running %s %s %s %s\n", argv[0], argv[1], argv[2], argv[3]);

  

    src_name = argv[1];

    dst_name = argv[2];

    block_size = atoi(argv[3]);

  

    if(block_size > MAX_BLOCK_SIZE) {

        printf("error: block size if too big (maximum is %d bytes (%d)\n", MAX_BLOCK_SIZE, __LINE__);

        return EXIT_FAILURE;

    }

  

    stat(src_name, &src_stat);

    file_size = src_stat.st_size;

    block_count = file_size / block_size;

    remainder = (size_t)(file_size % block_size);

  

    /* open source file */

    src_fp = fopen(src_name, "r");

    if(src_fp == NULL) {

        printf("error: %s (%d)\n", strerror(errno), __LINE__);

        return EXIT_FAILURE;

    }

  

    /* open destination file */

    dst_fp = fopen(dst_name, "w");

    if(dst_fp == NULL) {

        printf("error: %s (%d)\n", strerror(errno), __LINE__);

        return EXIT_FAILURE;

    }

  

    /* copy */

    for(i = 0; i < block_count; i++) {

        fread(buf, (size_t)block_size, 1, src_fp);

        fwrite(buf, (size_t)block_size, 1, dst_fp);

        copied += (size_t)block_size;

    }

  

    if(remainder > 0) {

        fread(buf, remainder, 1, src_fp);

        fwrite(buf, remainder, 1, dst_fp);

        copied += remainder;

    }

  

    /* close */

    fclose(src_fp);

    fclose(dst_fp);

  

    printf("%lu bytes copied\n", copied);

  

    return EXIT_SUCCESS;

}
