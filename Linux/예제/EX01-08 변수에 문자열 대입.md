# EX01-08 변수에 문자열 대입

```shell
#!/bin/bash

name="linux"
course="expert"

# linux expert
var="$name $course"
echo $var

# $linux \expert
var="\$$name \\$course"
echo $var

# $$linux \\expert
var="\$\$$name \\\\$course"
echo $var

# experted linux
var="${course}ed $name"
echo $var

exit 0
```
